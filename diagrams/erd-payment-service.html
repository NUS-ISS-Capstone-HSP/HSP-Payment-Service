    payments {
        string payment_id PK "UUID"
        string order_id FK "关联订单ID, UNIQUE"
        float amount "支付金额"
        enum method "WECHAT | CASH"
        enum status "PENDING | COMPLETED | FAILED | REFUNDED"
        datetime paid_at "支付完成时间"
        datetime created_at "创建时间"
        datetime updated_at "更新时间"
    }

    worker_incomes {
        string income_id PK "UUID"
        string order_id FK "关联订单ID"
        string worker_id FK "师傅ID"
        float order_amount "订单总金额"
        float commission_rate "平台抽成比例 (default 0.3)"
        float worker_amount "师傅所得金额"
        string settled "N=未结算 Y=已结算"
        datetime settled_at "结算时间"
        datetime created_at "创建时间"
    }

Table: payments

ColumnTypeConstraintDescription
payment_idVARCHAR(36)PRIMARY KEYUUID v4, 支付记录唯一标识
order_idVARCHAR(36)UNIQUE, INDEX关联订单ID (一对一)
amountFLOATNOT NULL支付金额 (元)

Table: worker_incomes

ColumnTypeConstraintDescription
income_idVARCHAR(36)PRIMARY KEYUUID v4, 收入记录唯一标识
order_idVARCHAR(36)INDEX关联订单ID
worker_idVARCHAR(36)INDEX师傅ID
order_amountFLOATNOT NULL订单总金额
commission_rateFLOATDEFAULT 0.7师傅分成比例 (70%)
worker_amountFLOATNOT NULL师傅所得 = order_amount × commission_rate
settledVARCHAR(1)DEFAULT 'N'N=未结算 / Y=已结算
settled_atDATETIMENULLABLE结算时间
created_atDATETIMEDEFAULT NOW()创建时间

Worker Income Calculation Formula

worker_amount = order_amount × commission_rate

Example: 订单金额 ¥500.00 × 70% 分成 = 师傅收入 ¥350.00 (平台抽成 ¥150.00)
